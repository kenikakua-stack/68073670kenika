  <div class="form-box">
    <h2>กรอกข้อมูลจัดส่ง</h2>
    <form id="orderForm">
      <input type="text" id="customer_name" placeholder="ชื่อ-นามสกุล ผู้รับ" required />
      <input type="tel" id="phone" placeholder="เบอร์โทรศัพท์" required />
      <textarea id="address" rows="3" placeholder="ที่อยู่จัดส่ง (บ้านเลขที่ ถนน ตำบล/แขวง อำเภอ/เขต จังหวัด รหัสไปรษณีย์)" required></textarea>
      <p id="summaryText">ยอดรวม: 0 บาท</p>
      <button type="submit" id="submitBtn">ยืนยันการสั่งซื้อ</button>
    </form>
    <p id="resultMsg"></p>
  </div>

  <script>
    const SCRIPT_URL = 'วาง_GOOGLE_APPS_SCRIPT_WEB_APP_URL_ที่นี่';
    let products = [];

    async function loadProducts() {
      const res = await fetch(`${SCRIPT_URL}?action=getProducts`);
      const data = await res.json();
      products = data.products || [];
      const container = document.getElementById('products');
      container.innerHTML = products.map((p, idx) => `
        <div class="card">
          <img src="${p.image}" alt="${p.name}" onerror="this.src='https://placehold.co/400x250?text=Honey+SKU'" />
          <h3>${p.name}</h3>
          <small>รหัส SKU: ${p.sku} | คงเหลือ: ${p.stock} ขวด</small>
          <div class="price">${Number(p.price).toLocaleString()} บาท</div>
          <label>
            จำนวน:
            <input type="number" min="0" max="${p.stock}" value="0" data-idx="${idx}" class="qty-input" onchange="calcTotal()" />
          </label>
        </div>
      `).join('');
    }

    function getSelectedItems() {
      const inputs = document.querySelectorAll('.qty-input');
      const selected = [];
      inputs.forEach(input => {
        const qty = Number(input.value || 0);
        if (qty > 0) {
          const p = products[Number(input.dataset.idx)];
          selected.push({ sku: p.sku, name: p.name, price: Number(p.price), qty });
        }
      });
      return selected;
    }

    function calcTotal() {
      const items = getSelectedItems();
      const total = items.reduce((sum, item) => sum + item.price * item.qty, 0);
      document.getElementById('summaryText').textContent = `ยอดรวม: ${total.toLocaleString()} บาท`;
      return total;
    }

    document.getElementById('orderForm').addEventListener('submit', async (e) => {
      e.preventDefault();
      const items = getSelectedItems();
      if (items.length === 0) return alert('กรุณาเลือกสินค้าอย่างน้อย 1 ชิ้น');

      const btn = document.getElementById('submitBtn');
      btn.disabled = true;
      btn.textContent = 'กำลังบันทึกคำสั่งซื้อ...';

      const payload = {
        action: 'createOrder',
        customer_name: document.getElementById('customer_name').value.trim(),
        phone: document.getElementById('phone').value.trim(),
        address: document.getElementById('address').value.trim(),
        items,
        items_summary: items.map(i => `${i.name} (${i.sku}) x${i.qty}`).join(', '),
        total: calcTotal()
      };

      const res = await fetch(SCRIPT_URL, {
        method: 'POST',
        body: JSON.stringify(payload)
      });
      const result = await res.json();

      document.getElementById('resultMsg').textContent =
        result.status === 'ok'
          ? `สั่งซื้อสำเร็จ! หมายเลขออเดอร์ของคุณคือ ${result.orderId}`
          : 'เกิดข้อผิดพลาด กรุณาลองใหม่อีกครั้ง';

      btn.disabled = false;
      btn.textContent = 'ยืนยันการสั่งซื้อ';
      e.target.reset();
      calcTotal();
      loadProducts();
    });

    loadProducts();
  </script>
